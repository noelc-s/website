  function createRow() {
    const row = document.createElement('div');
    row.className = 'row';
    return row;
  }

  function chunkArray(arr, size) {
    const chunks = [];
    for (let i = 0; i < arr.length; i += size) {
      chunks.push(arr.slice(i, i + size));
    }
    return chunks;
  }

  function renderAlternatingRows(horizontalImages, verticalImages, containerId) {
    const gallery = document.getElementById(containerId);
    gallery.innerHTML = '';

    const horizontalChunks = chunkArray(horizontalImages, 2);
    const verticalChunks = chunkArray(verticalImages, 3);

    const maxRows = Math.max(horizontalChunks.length, verticalChunks.length);
    let hIndex = 0, vIndex = 0;

    for (let i = 0; i < maxRows * 2; i++) {
      const isHorizontalRow = (i % 2 === 0);
      const row = createRow();

      if (isHorizontalRow && hIndex < horizontalChunks.length) {
        horizontalChunks[hIndex].forEach(img => {
          const col = document.createElement('div');
          col.className = 'column-horizontal';

          const wrapper = document.createElement('div');
          wrapper.className = 'image-wrapper-horizontal';

          const image = document.createElement('img');
          image.src = img;
          image.alt = "";
          image.className = 'rounded-corners';

          wrapper.appendChild(image);
          col.appendChild(wrapper);
          row.appendChild(col);
        });
        hIndex++;
      } else if (!isHorizontalRow && vIndex < verticalChunks.length) {
        verticalChunks[vIndex].forEach(img => {
          const col = document.createElement('div');
          col.className = 'column-vertical';

          const wrapper = document.createElement('div');
          wrapper.className = 'image-wrapper-vertical';

          const image = document.createElement('img');
          image.src = img;
          image.alt = "";
          image.className = 'rounded-corners';

          wrapper.appendChild(image);
          col.appendChild(wrapper);
          row.appendChild(col);
        });
        vIndex++;
      }
      gallery.appendChild(row);
    }
  }

  // Determine orientation client-side
  const promises = images.map(filename => {
    return new Promise(resolve => {
      const img = new Image();
      img.onload = () => {
        resolve({
          src: imageFolder + filename,
          isHorizontal: img.naturalWidth >= img.naturalHeight
        });
      };
      img.src = imageFolder + filename;
    });
  });

  Promise.all(promises).then(files => {
    const horizontal = files.filter(f => f.isHorizontal).map(f => f.src);
    const vertical = files.filter(f => !f.isHorizontal).map(f => f.src);
    renderAlternatingRows(horizontal, vertical, 'gallery-container');
  });
