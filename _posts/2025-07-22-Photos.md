  const folderPath = '/Photos/Best';

  function createRow() {
    const row = document.createElement('div');
    row.className = 'row';
    return row;
  }

  function chunkArray(arr, size) {
    const chunks = [];
    for (let i = 0; i < arr.length; i += size) {
      chunks.push(arr.slice(i, i + size));
    }
    return chunks;
  }

  // Render rows alternating horizontal and vertical
  function renderRowsAlternating(horizontalChunks, verticalChunks, gallery) {
    let hIndex = 0, vIndex = 0;
    const maxRows = Math.max(horizontalChunks.length, verticalChunks.length);
    for (let i = 0; i < maxRows * 2; i++) {
      const isHorizontal = i % 2 === 0;
      let rowImages = null;
      if (isHorizontal) {
        if (hIndex >= horizontalChunks.length) continue;
        rowImages = horizontalChunks[hIndex++];
      } else {
        if (vIndex >= verticalChunks.length) continue;
        rowImages = verticalChunks[vIndex++];
      }

      const row = createRow();

      rowImages.forEach(imgData => {
        const col = document.createElement('div');
        col.className = isHorizontal ? 'column-horizontal' : 'column-vertical';

        const container = document.createElement('div');
        container.className = isHorizontal ? 'container-horizontal' : 'container-vertical';

        const img = document.createElement('img');
        img.className = 'rounded-corners';
        img.alt = imgData.name;

        container.appendChild(img);
        col.appendChild(container);
        row.appendChild(col);

        // Load image dynamically
        img.onload = () => {
          // image loaded — can do something here if needed
        };
        img.onerror = () => {
          console.error('Image failed to load:', imgData.name);
        };
        img.src = imgData.url;
      });

      gallery.appendChild(row);
    }
  }

  async function getTempLinkWithDelay(path, delayMs = 300) {
    const res = await fetch('https://api.dropboxapi.com/2/files/get_temporary_link', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${accessToken}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ path })
    });
    const json = await res.json();
    await new Promise(resolve => setTimeout(resolve, delayMs)); // delay to avoid rate limits
    return json.link;
  }

  async function fetchAndRender() {
    const gallery = document.getElementById('gallery-container');
    gallery.innerHTML = '';

    const listRes = await fetch('https://api.dropboxapi.com/2/files/list_folder', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${accessToken}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ path: folderPath })
    });
    const data = await listRes.json();
    const files = data.entries.filter(entry =>
      entry['.tag'] === 'file' && /\.(jpe?g|png|gif)$/i.test(entry.name)
    );

    // We will collect images as we go, and render rows incrementally
    const horizontalImages = [];
    const verticalImages = [];

    // Chunk size for rows
    const horizontalChunkSize = 2;
    const verticalChunkSize = 3;

    // For progressive rendering, create a function that tries to render rows as soon as possible
    function tryRender() {
      const horizontalChunks = chunkArray(horizontalImages, horizontalChunkSize);
      const verticalChunks = chunkArray(verticalImages, verticalChunkSize);

      gallery.innerHTML = ''; // clear

      renderRowsAlternating(horizontalChunks, verticalChunks, gallery);
    }

    for (const file of files) {
      try {
        const url = await getTempLinkWithDelay(file.path_lower, 300);

        await new Promise((resolve) => {
          const img = new Image();
          img.onload = () => {
            const isHorizontal = img.naturalWidth >= img.naturalHeight;
            const imgData = {
              name: file.name,
              url,
              width: img.naturalWidth,
              height: img.naturalHeight,
              isHorizontal,
            };
            if (isHorizontal) horizontalImages.push(imgData);
            else verticalImages.push(imgData);

            tryRender();
            resolve();
          };
          img.onerror = () => {
            console.error('Failed to load image:', file.name);
            resolve();
          };
          img.src = url;
        });
      } catch (err) {
        console.error('Error loading image:', file.name, err);
      }
    }
  }

  fetchAndRender().catch(err => {
    console.error('Error fetching photos:', err);
  });
